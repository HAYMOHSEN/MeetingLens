cmake_minimum_required(VERSION 3.20)
project(MeetingLensBasicCamera LANGUAGES CXX)
if(NOT WIN32)
  message(FATAL_ERROR "BasicCamera uses Windows DirectShow and must be built with the Windows SDK.")
endif()
add_executable(MeetingLens.BasicCamera BasicCamera.cpp)
target_compile_features(MeetingLens.BasicCamera PRIVATE cxx_std_17)
target_compile_definitions(MeetingLens.BasicCamera PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
target_link_libraries(MeetingLens.BasicCamera PRIVATE strmiids ole32 oleaut32)
if(MSVC)
  target_compile_options(MeetingLens.BasicCamera PRIVATE /W4 /WX /EHsc)
endif()
